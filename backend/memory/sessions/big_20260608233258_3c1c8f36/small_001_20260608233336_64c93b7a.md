#### 3.2 事件研究法模块 (`analysis/event_study.py`)
```python
import numpy as np
from scipy import stats

class EventStudy:
    """事件研究法引擎，用于量化特朗普访华影响"""
    
    def __init__(self, market_data, event_date='2026-05-01'):
        self.market_data = market_data
        self.event_date = pd.Timestamp(event_date)
        self.estimation_window = 120  # 估计窗口120天
        self.event_window = (-30, 60) # 事件窗口[-30, +60]天
        
    def calculate_abnormal_returns(self, asset='SPY'):
        """使用市场模型计算异常收益率 (AR/CAR)"""
        # 估计窗口：[-150, -30]
        est_start = self.event_date - pd.Timedelta(days=150)
        est_end = self.event_date - pd.Timedelta(days=30)
        
        # 市场模型: R_i = alpha + beta * R_m + epsilon
        asset_returns = self.market_data[asset].pct_change()
        market_returns = self.market_data['SPY'].pct_change()  # 使用SPY作为市场代理
        
        # 估计alpha, beta
        idx = self.market_data.index.slice_indexer(est_start, est_end)
        X = market_returns[idx][1:].values.reshape(-1, 1)
        y = asset_returns[idx][1:].values
        beta, alpha = np.polyfit(X.flatten(), y, 1)
        
        # 计算事件窗口AR和CAR
        event_start = self.event_date - pd.Timedelta(days=30)
        event_end = self.event_date + pd.Timedelta(days=60)
        idx_e = self.market_data.index.slice_indexer(event_start, event_end)
        
        expected_returns = alpha + beta * market_returns[idx_e][1:]
        actual_returns = asset_returns[idx_e][1:]
        abnormal_returns = actual_returns - expected_returns
        cumulative_ar = abnormal_returns.cumsum()
        
        return {
            'AR': abnormal_returns,      # 异常收益率
            'CAR': cumulative_ar,        # 累积异常收益率
            'beta': beta,
            'alpha': alpha,
            't_stat': np.mean(abnormal_returns) / (np.std(abnormal_returns) / np.sqrt(len(abnormal_returns)))
        }
    
    def compare_with_history(self):
        """与历史领导人互访事件对比（2017年特朗普访华、2023年习拜会）"""
        history_events = {
            '2017-11-08': 'Trump_China_2017',
            '2023-11-15': 'Xi_Biden_2023',
            '2025-10-30': 'Trump_Xi_2025'
        }
        results = {}
        for date, label in history_events.items():
            self.event_date = pd.Timestamp(date)
            ar_result = self.calculate_abnormal_returns()
            results[label] = ar_result['CAR']
        return pd.DataFrame(results)
```
