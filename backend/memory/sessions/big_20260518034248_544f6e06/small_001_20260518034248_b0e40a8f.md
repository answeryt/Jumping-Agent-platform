#!/usr/bin/env python3
"""
中国领导人外事接待活动自动检索脚本

根据调研报告，从新华网时政频道抓取最新新闻，通过关键词筛选出与
"外国领导人访华/会见"相关的报道，并结构化输出。

依赖：requests, beautifulsoup4, lxml
安装：pip install requests beautifulsoup4 lxml
"""

import requests
from bs4 import BeautifulSoup
import re
from datetime import datetime
from typing import List, Dict, Optional

# ========== 配置区 ==========
TARGET_URL = "http://www.xinhuanet.com/politics/"
HEADERS = {
    "User-Agent": (
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
        "AppleWebKit/537.36 (KHTML, like Gecko) "
        "Chrome/120.0.0.0 Safari/537.36"
    ),
    "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
    "Accept-Language": "zh-CN,zh;q=0.9,en;q=0.8",
}

# 用于判断外事领导人接待活动的关键词（支持大小写）
KEYWORDS = [
    "会见", "访华", "会谈", "对话", "接见",
    "总统", "总理", "首相", "主席", "国王", "总督",
    "外长", "外交部长", "国事访问",
    "习近平", "李强", "赵乐际", "王沪宁", "蔡奇",
    "丁薛祥", "李希", "韩正",
    "王毅",
]

# ========== 核心功能 ==========

def fetch_xinhuanet_politics(url: str = TARGET_URL) -> Optional[str]:
    """
    抓取新华网时政频道的 HTML 内容。

    Args:
        url: 目标 URL，默认新华网时政频道。

    Returns:
        成功返回 HTML 字符串，失败返回 None。
    """
    try:
        response = requests.get(url, headers=HEADERS, timeout=15)
        response.raise_for_status()
        # 确认页面编码
        response.encoding = response.apparent_encoding or "utf-8"
        return response.text
    except requests.exceptions.RequestException as e:
        print(f"[Error] 网络请求失败: {e}")
        return None


def parse_articles(html: str) -> List[Dict[str, str]]:
    """
    解析 HTML 页面，提取文章列表。

    Args:
        html: 原始 HTML 字符串。

    Returns:
        文章字典列表，每项包含 title, url, time, summary。
        若提取失败则返回空列表。
    """
    soup = BeautifulSoup(html, "lxml")
    articles = []

    # 尝试多种常见的新闻列表容器选择器（适配不同页面结构）
    # 优先级：data-driven 模式 > 常规 class 模式
    article_items = []

    # 1. 尝试匹配 <a> 标签嵌套的新闻条目
    for a_tag in soup.find_all("a", href=True):
        href = a_tag.get("href", "")
        text = a_tag.get_text(strip=True)
        # 至少包含一定有效文字且链接看起来像新闻内容页
        if len(text) > 10 and ("content" in href or "xinhuanet" in href or href.startswith("/")):
            article_items.append(a_tag)

    # 2. 如果上面太粗糙，尝试更精确的选择器（针对新华网常用结构）
    for selector in [
        "div.part-hd a",           # 头部标题类
        "h3 a",                    # 多级的 h3
        "div.news-list li a",      # 列表项
        "ul li a[href*='xinhuanet']",
    ]:
        items = soup.select(selector)
        if items:
            article_items = items
            break

    # 如果还是没找到，回退到全部 a 标签（极端情况）
    if not article_items:
        article_items = soup.find_all("a", href=True)

    seen_urls = set()
    for item in article_items:
        title = item.get_text(strip=True)
        url = item.get("href", "")

        # 过滤无效项
        if not title or len(title) < 8:
            continue

        # 相对路径转绝对路径
        if url.startswith("/"):
            url = "http://www.xinhuanet.com" + url
        elif url.startswith("//"):
            url = "https:" + url
        elif not url.startswith("http"):
            continue  # 跳过 javascript: 等无效链接

        # 去重
        if url in seen_urls:
            continue
        seen_urls.add(url)

        # 尝试提取时间：从页面中查找最近的 <span class="time"> 或父级元素
        time_str = ""
        parent = item.find_parent("li") or item.find_parent("div")
        if parent:
            time_tag = parent.find("span", class_=re.compile(r"time", re.I))
            if time_tag:
                time_str = time_tag.get_text(strip=True)
            else:
                # 尝试从链接附近文本提取日期（如"2025-07-15"）
                all_text = parent.get_text()
                match = re.search(r"(\d{4}[-/]\d{1,2}[-/]\d{1,2})", all_text)
                if match:
                    time_str = match.group(1)

        # 提取摘要：找同级或父级下的 p / span.desc / div.desc
        summary = ""
        if parent:
            desc = parent.find("p") or parent.find(class_=re.compile(r"desc|summary", re.I))
            if desc:
                summary = desc.get_text(strip=True)[:200]  # 截取前200字符

        articles.append({
            "title": title,
            "url": url,
            "time": time_str,
            "summary": summary,
        })

    return articles


def filter_foreign_leader_articles(articles: List[Dict[str, str]]) -> List[Dict[str, str]]:
    """
    通过关键词过滤与外国领导人接待相关的文章。

    Args:
        articles: 文章字典列表。

    Returns:
        过滤后的文章列表。
    """
    filtered = []
    for article in articles:
        combined_text = f"{article['title']} {article['summary']}"
        # 检查是否包含至少两条关键词（或一条强关联词）
        matched = [kw for kw in KEYWORDS if kw in combined_text]
        if matched:
            article["matched_keywords"] = matched
            filtered.append(article)
    return filtered


def format_output(articles: List[Dict[str, str]]) -> str:
    """
    将文章列表格式化为易读的文本报告。

    Args:
        articles: 过滤后的文章列表。

    Returns:
        格式化后的字符串。
    """
    if not articles:
        return "未找到与外国领导人访华/会见相关的新闻报道。"

    lines = ["===== 中国近期接待外国领导人相关报道 =====\n"]
    lines.append(f"共找到 {len(articles)} 条相关新闻\n")

    for idx, article in enumerate(articles, 1):
        title = article["title"]
        time_str = article.get("time", "时间未知")
        summary = article.get("summary", "")
        url = article.get("url", "")
        kw = article.get("matched_keywords", [])

        lines.append(f"{idx}. [{time_str}] {title}")
        if summary:
            lines.append(f"   摘要：{summary}")
        lines.append(f"   链接：{url}")
        if kw:
            lines.append(f"   匹配关键词：{', '.join(kw)}")
        lines.append("")  # 空行分隔

    return "\n".join(lines)
